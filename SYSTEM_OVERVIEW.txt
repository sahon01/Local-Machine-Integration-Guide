┌────────────────────────────────────────────────────────────────────────────────────────┐
│                        🔐 API ENDPOINTS SUMMARY                                        │
└────────────────────────────────────────────────────────────────────────────────────────┘

  AUTHENTICATION
  ├─ POST   /api/auth/login                    → User login
  ├─ POST   /api/auth/register                 → User registration
  ├─ POST   /api/auth/logout                   → Logout
  └─ GET    /api/auth/me                       → Get user profile

  CHAT & COMPLETIONS
  ├─ POST   /api/completions/chat/completions → Chat completion (OpenAI compatible)
  ├─ GET    /api/completions/models            → List available models
  ├─ GET    /api/completions/models/:id        → Get specific model
  └─ POST   /api/completions/embeddings        → Generate embeddings

  PROVIDERS
  ├─ GET    /api/providers                     → List providers
  ├─ GET    /api/providers/:id                 → Get provider details
  ├─ POST   /api/providers                     → Add provider
  ├─ PUT    /api/providers/:id                 → Update provider
  └─ DELETE /api/providers/:id                 → Delete provider

  MODELS
  ├─ GET    /api/models                        → List models
  ├─ GET    /api/models/:id                    → Get model details
  ├─ POST   /api/models                        → Add model
  ├─ PUT    /api/models/:id                    → Update model
  └─ DELETE /api/models/:id                    → Delete model

  AGENTS
  ├─ GET    /api/agents                        → List agents
  ├─ GET    /api/agents/:id                    → Get agent details
  ├─ POST   /api/agents                        → Create agent
  ├─ PUT    /api/agents/:id                    → Update agent
  └─ DELETE /api/agents/:id                    → Delete agent

  SERVERS
  ├─ GET    /api/servers                       → List servers
  ├─ GET    /api/servers/:id                   → Get server details
  ├─ POST   /api/servers                       → Add server
  ├─ PUT    /api/servers/:id                   → Update server
  └─ DELETE /api/servers/:id                   → Delete server

  TOOLS
  ├─ GET    /api/tools                         → List tools
  ├─ GET    /api/tools/:id                     → Get tool details
  ├─ POST   /api/tools                         → Create tool
  ├─ PUT    /api/tools/:id                     → Update tool
  └─ DELETE /api/tools/:id                     → Delete tool

  ADMIN
  ├─ GET    /api/admin/stats                   → System statistics
  ├─ GET    /api/admin/health/all              → System health check
  ├─ GET    /api/admin/config                  → System configuration
  └─ PUT    /api/admin/config                  → Update configuration


┌────────────────────────────────────────────────────────────────────────────────────────┐
│                        📊 DATABASE SCHEMA (11 Tables)                                   │
└────────────────────────────────────────────────────────────────────────────────────────┘

  USERS TABLE
  ├─ id (UUID)
  ├─ email (unique)
  ├─ username (unique)
  ├─ password_hash
  ├─ role (admin|user|guest)
  ├─ status (active|inactive|suspended)
  ├─ last_login, created_at, updated_at

  PROVIDERS TABLE
  ├─ id (UUID)
  ├─ name, type (openai|ollama|gemini|custom)
  ├─ api_url, api_key
  ├─ is_active, is_default
  ├─ rate_limit_requests, rate_limit_period
  └─ created_at, updated_at

  MODELS TABLE
  ├─ id (UUID)
  ├─ provider_id (FK)
  ├─ model_id (e.g., gpt-4)
  ├─ name, display_name
  ├─ context_window, max_tokens
  ├─ is_active, is_default
  └─ created_at, updated_at

  AGENTS TABLE
  ├─ id (UUID)
  ├─ name, description
  ├─ type (general|code_editor|chatting|master|custom)
  ├─ model_id, server_id (FK)
  ├─ system_prompt
  ├─ status (idle|active|error|training)
  ├─ temperature, top_p, max_tokens
  └─ created_at, updated_at

  SERVERS TABLE
  ├─ id (UUID)
  ├─ hostname, ip_address, port
  ├─ region, os
  ├─ status (online|offline|maintenance)
  ├─ cpu_usage, memory_usage, disk_usage
  └─ last_health_check

  TOOLS TABLE
  ├─ id (UUID)
  ├─ name, description
  ├─ type (admin|agent|tool)
  ├─ is_active
  └─ configuration (JSON)

  RAG_DOCUMENTS TABLE
  ├─ id (UUID)
  ├─ title, content
  ├─ embedding (vector)
  ├─ metadata (JSON)
  └─ created_at

  CONVERSATION_MEMORY TABLE
  ├─ id (UUID)
  ├─ agent_id, user_id (FK)
  ├─ messages (JSON)
  ├─ context (JSON)
  └─ created_at, updated_at

  API_REQUESTS TABLE
  ├─ id (UUID)
  ├─ user_id, provider_id, model_id (FK)
  ├─ request_type
  ├─ tokens_used, response_time
  ├─ status_code, error_message
  └─ created_at

  SETTINGS TABLE
  ├─ id (UUID)
  ├─ key, value
  ├─ category, description
  └─ updated_at

  WEBHOOKS TABLE
  ├─ id (UUID)
  ├─ event_type, url
  ├─ is_active, secret_key
  └─ created_at
